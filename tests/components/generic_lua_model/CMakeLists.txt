macro(gs_add_test test)
    add_executable(${test} ${test}.cc)
     target_link_libraries(${test} PRIVATE gtest gmock router gs_memory reg_router generic_lua_model qbox-common)
    add_test(NAME ${test} COMMAND ${test} --gs_luafile ${CMAKE_CURRENT_SOURCE_DIR}/conf.lua)
    set_tests_properties(${test} PROPERTIES TIMEOUT 10)
endmacro()
gs_add_test(generic_lua_model-tests)
